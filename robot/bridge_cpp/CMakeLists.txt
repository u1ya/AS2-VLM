cmake_minimum_required(VERSION 3.16)
project(as2_vlm_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(unitree_sdk2 REQUIRED)
add_executable(as2_state_probe state_probe.cpp)
target_link_libraries(as2_state_probe PRIVATE unitree_sdk2)
install(TARGETS as2_state_probe RUNTIME DESTINATION bin)
add_executable(as2_stop_only stop_only.cpp)
target_link_libraries(as2_stop_only PRIVATE unitree_sdk2)
install(TARGETS as2_stop_only RUNTIME DESTINATION bin)
add_executable(as2_forward_pulse forward_pulse.cpp)
target_link_libraries(as2_forward_pulse PRIVATE unitree_sdk2)
add_executable(as2_forward_pulse_test forward_pulse_test.cpp)
enable_testing()
add_test(NAME forward_pulse_policy COMMAND as2_forward_pulse_test)
add_executable(as2_walk_meter walk_meter.cpp)
target_link_libraries(as2_walk_meter PRIVATE unitree_sdk2)
add_executable(as2_meter_policy_test meter_policy_test.cpp)
add_test(NAME meter_policy COMMAND as2_meter_policy_test)
add_executable(as2_camera_probe camera_probe.cpp)
target_link_libraries(as2_camera_probe PRIVATE unitree_sdk2)
add_test(NAME camera_probe_help COMMAND as2_camera_probe --help)
add_test(NAME camera_probe_invalid_service COMMAND as2_camera_probe unused invalid unused.jpg)
set_tests_properties(camera_probe_invalid_service PROPERTIES WILL_FAIL TRUE)
add_executable(as2_dds_discover dds_discover.cpp)
target_link_libraries(as2_dds_discover PRIVATE unitree_sdk2)
add_test(NAME dds_discover_help COMMAND as2_dds_discover --help)
add_executable(as2_lidar_probe lidar_probe.cpp)
target_link_libraries(as2_lidar_probe PRIVATE unitree_sdk2)
add_test(NAME lidar_probe_help COMMAND as2_lidar_probe --help)
